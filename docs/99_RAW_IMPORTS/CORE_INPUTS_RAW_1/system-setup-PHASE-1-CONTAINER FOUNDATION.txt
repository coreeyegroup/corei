	3. EXECUTION STEPS (DETAILED)
		STEP 1.1 — CLEAN PREVIOUS DOCKER INSTALLATIONS

			Action

				Removed any existing Docker packages

				sudo apt remove -y docker docker-engine docker.io containerd runc

			Outcome

				Clean environment ensured (no conflicting binaries)

		STEP 1.2 — INSTALL SYSTEM DEPENDENCIES
				sudo apt update
				sudo apt install -y ca-certificates curl gnupg lsb-release

			Outcome

				System prepared for secure repository integration

		STEP 1.3 — ADD DOCKER GPG KEY
				sudo mkdir -p /etc/apt/keyrings

				curl -fsSL https://download.docker.com/linux/ubuntu/gpg | \
				sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg

			Outcome

				Trusted package verification enabled

		STEP 1.4 — ADD OFFICIAL DOCKER REPOSITORY
				echo \
				"deb [arch=$(dpkg --print-architecture) \
				signed-by=/etc/apt/keyrings/docker.gpg] \
				https://download.docker.com/linux/ubuntu \
				$(lsb_release -cs) stable" | \
				sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

			Outcome

				Official Docker repository configured

		STEP 1.5 — INSTALL DOCKER ENGINE
				sudo apt update

				sudo apt install -y docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin

			Outcome

				Docker Engine + CLI + container runtime installed

		STEP 1.6 — ENABLE & START DOCKER SERVICE
				sudo systemctl enable docker
				sudo systemctl start docker

			Outcome

				Docker auto-start enabled
				Service running

		STEP 1.7 — VERIFY DOCKER SERVICE
				sudo systemctl status docker

			Result

				Active: active (running)

			Outcome

				Docker daemon operational

		STEP 1.8 — CONFIGURE NON-ROOT USER ACCESS

			Outcome
