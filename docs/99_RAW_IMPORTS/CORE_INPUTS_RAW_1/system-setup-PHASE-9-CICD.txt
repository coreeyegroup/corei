					data:
					  deploy.sh: |
						#!/bin/sh

						echo "Starting SAFE Deploy Agent..."

						IMAGE_TAG=$(kubectl get configmap sample-service-image -n platform -o jsonpath="{.data.IMAGE_TAG}")
						PREVIOUS_TAG=$(kubectl get configmap sample-service-image -n platform -o jsonpath="{.data.PREVIOUS_TAG}")

						IMAGE="coreeyehub/sample-service:$IMAGE_TAG"

						kubectl set image deployment/sample-service \
						sample-service=$IMAGE \
						-n platform

						kubectl rollout status deployment/sample-service -n platform --timeout=60s

						if [ $? -ne 0 ]; then
						  echo "ROLLBACK TRIGGERED"

						  kubectl set image deployment/sample-service \
						  sample-service=coreeyehub/sample-service:$PREVIOUS_TAG \
						  -n platform

						  exit 1
						fi

						kubectl patch configmap sample-service-image \
						-n platform \
						-p "{\"data\":{\"PREVIOUS_TAG\":\"$IMAGE_TAG\"}}"

						echo "SAFE DEPLOY COMPLETED"

				#!/bin/bash

				set -e

				REGISTRY="localhost:5000/core"
				VERSION_FILE=~/core-system/platform/versioning/version.txt

				get_version() {
				  SERVICE=$1
				  grep "^$SERVICE=" $VERSION_FILE | cut -d'=' -f2
				}

				SERVICES=(
				  "signal-service"
				)

				for SERVICE in "${SERVICES[@]}"
				do
				  VERSION=$(get_version $SERVICE)

				  if [ -z "$VERSION" ]; then
					echo "ERROR: Version not found for $SERVICE"
					exit 1
				  fi

				  echo "Checking existing versions..."

				  if curl -s http://localhost:5000/v2/core/$SERVICE/tags/list | grep -q "$VERSION"; then
					echo "ERROR: Version $VERSION already exists for $SERVICE"
					exit 1
				  fi

				  echo "--------------------------------------"
				  echo "Building $SERVICE:$VERSION"
				  echo "--------------------------------------"

				  docker build --no-cache -t $REGISTRY/$SERVICE:$VERSION \
					~/core-system/services/$SERVICE

				  echo "Pushing $SERVICE:$VERSION"

				  docker push $REGISTRY/$SERVICE:$VERSION

				done

				echo "--------------------------------------"
				echo "VERSIONED BUILD COMPLETE"
				echo "--------------------------------------"
