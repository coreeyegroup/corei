	5. REGISTRY HARDENING
		🔧 STEP 1 — RETAG IMAGE
			docker tag localhost:5000/alertmanager-telegram:v2 192.168.1.3:5000/alertmanager-telegram:v2

			FILE

				{
				  "insecure-registries": ["192.168.1.3:5000"]
				}

			🔧 COMMANDS
				sudo systemctl restart docker
				docker info | grep -i insecure

		🔧 STEP 3 — CONFIGURE CONTAINERD
			FILE
				/etc/containerd/config.toml

			🔴 IMPORTANT FIX

			Correct section:

				[plugins.'io.containerd.cri.v1.images'.registry]
				  config_path = ''

				  [plugins.'io.containerd.cri.v1.images'.registry.mirrors."192.168.1.3:5000"]
					endpoint = ["http://192.168.1.3:5000"]

				  [plugins.'io.containerd.cri.v1.images'.registry.configs."192.168.1.3:5000".tls]
					insecure_skip_verify = true

			🔧 COMMANDS
				sudo systemctl restart containerd
				kubectl delete pod -l app=alertmanager-telegram

				ImagePullBackOff → RESOLVED

	6. SIGNAL SERVICE FAILURE
		❗ ERROR
			CrashLoopBackOff
			Exit Code: 137
			Readiness probe failed

		ROOT CAUSE
			Worker service treated as HTTP service

		🔧 FIX
			COMMAND

				kubectl patch deployment signal-service \
				-p '{
				  "spec": {
					"template": {
					  "spec": {
						"containers": [
						  {
							"name": "signal-service",
							"ports": [],
							"livenessProbe": null,
							"readinessProbe": null
						  }
						]
					  }
					}
				  }
				}'

			🔁 RESTART
				kubectl rollout restart deployment signal-service

			✅ RESULT
				CrashLoopBackOff → RESOLVED

PHASE 10 — STEP 2 - ENVIRONMENT STRUCTURE (DEV / STAGING / PROD)

	OBJECTIVE

		Transform deployment model from:

		single static deployment per service

		to:

		multi-environment, isolated, Helm-controlled deployment architecture

	FINAL OUTCOME:

		Environment-based deployments:

			decision-engine-dev
			risk-engine-dev
			event-service-dev

		Each environment:

			independently deployable
			independently scalable
			isolated via Helm release
		configurable via values files

	1. ARCHITECTURE DESIGN
		CORE PRINCIPLE

			Same Helm chart + different values + different release name = isolated environment

		ENVIRONMENT MODEL

		Per service:

			decision-engine-dev
			decision-engine-staging
			decision-engine-prod

			risk-engine-dev
			risk-engine-staging
		risk-engine-prod

			event-service-dev
			event-service-staging
			event-service-prod

			ISOLATION MECHANISM
			Layer				Mechanism
			Deployment			Helm release name
			Configuration		values-*.yaml
			Scaling				replicaCount
			Resources			CPU / Memory
			Naming				{{ .Release.Name }}

	2. FILE STRUCTURE
		PATH (EXAMPLE)
			~/core-system/services/<service>/helm/<service>/

		REQUIRED FILES
			values-dev.yaml
			values-staging.yaml
			values-prod.yaml

			templates/
			  deployment.yaml
			  service.yaml

	3. VALUES FILES (PER SERVICE)
		values-dev.yaml
			Example (decision-engine):

			image:
			  repository: 192.168.1.3:5000/decision-engine
			  tag: latest

			replicaCount: 1

			resources:
			  limits:
				cpu: "200m"
				memory: "256Mi"

			env:
			  ENVIRONMENT: "dev"

		values-staging.yaml

			image:
			  repository: 192.168.1.3:5000/decision-engine
			  tag: latest

			replicaCount: 2

			resources:
			  limits:
				cpu: "300m"
				memory: "512Mi"

			env:
			  ENVIRONMENT: "staging"

		values-prod.yaml

			image:
			  repository: 192.168.1.3:5000/decision-engine
			  tag: latest

			replicaCount: 3

			resources:
			  limits:
				cpu: "500m"
				memory: "1Gi"

			env:
			  ENVIRONMENT: "prod"

	4. DEPLOYMENT TEMPLATE STANDARDIZATION
		FILE
			templates/deployment.yaml

		FINAL STRUCTURE:

			apiVersion: apps/v1
			kind: Deployment
			metadata:
			  name: {{ .Release.Name }}
			  labels:
				app: {{ .Release.Name }}

			spec:
			  replicas: {{ .Values.replicaCount }}

			  selector:
				matchLabels:
				  app: {{ .Release.Name }}

			  template:
				metadata:
				  labels:
					app: {{ .Release.Name }}

				spec:
				  containers:
					- name: {{ .Release.Name }}

					  image: "{{ .Values.image.repository }}:{{ .Values.image.tag }}"

					  ports:
						- containerPort: 8000

					  env:
						- name: ENVIRONMENT
						  value: "{{ .Values.env.ENVIRONMENT }}"

					  resources:
						limits:
						  cpu: "{{ .Values.resources.limits.cpu }}"
						  memory: "{{ .Values.resources.limits.memory }}"

		KEY CHANGES
		Static:
			name: decision-engine
			replicas: 1

		Replaced with:
			name: {{ .Release.Name }}
			replicas: {{ .Values.replicaCount }}

	5. SERVICE TEMPLATE STANDARDIZATION
		FILE
			templates/service.yaml

		FINAL STRUCTURE
			apiVersion: v1
			kind: Service
			metadata:
			  name: {{ .Release.Name }}
			  labels:
				app: {{ .Release.Name }}

			spec:
			  selector:
				app: {{ .Release.Name }}

			  ports:
				- port: 8000
				  targetPort: 8000

			CRITICAL RULE:
			All Kubernetes resources must use:
				{{ .Release.Name }}
