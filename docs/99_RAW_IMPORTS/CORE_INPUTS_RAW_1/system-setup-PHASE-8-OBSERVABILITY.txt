			5.4 LAG CALCULATION FUNCTION

				def calculate_lag():
					consumer = KafkaConsumer(
						bootstrap_servers=KAFKA_BOOTSTRAP,
						group_id=GROUP_ID,
						enable_auto_commit=False
					)

					partitions = consumer.partitions_for_topic(TOPIC)
					topic_partitions = [TopicPartition(TOPIC, p) for p in partitions]

					consumer.assign(topic_partitions)

					end_offsets = consumer.end_offsets(topic_partitions)
					committed_offsets = {
						tp: consumer.committed(tp) for tp in topic_partitions
					}

					total_lag = 0

					for tp in topic_partitions:
						end = end_offsets[tp]
						committed = committed_offsets[tp] or 0
						lag = end - committed
						total_lag += lag

					consumer.close()
					return total_lag

			5.5 PROMETHEUS METRIC
				Metric Definition

					from prometheus_client import Gauge

					consumer_lag_gauge = Gauge(
						"consumer_lag",
						"Total Kafka consumer lag"
					)

			5.6 BACKGROUND LOOP (CRITICAL)

				import time
				import threading

				def lag_monitor():
					while True:
						lag = calculate_lag()
						consumer_lag_gauge.set(lag)
						time.sleep(5)

				threading.Thread(target=lag_monitor, daemon=True).start()

	5. 📉 LAG MONITOR ENGINE (CRITICAL)

			import time
			from kafka import KafkaConsumer, TopicPartition
			from app.metrics import consumer_lag

			KAFKA_BROKER = "kafka.kafka.svc.cluster.local:9092"
			TOPIC = "test-topic"
			GROUP_ID = "metrics-consumer-group"

			def monitor_lag():
				consumer = KafkaConsumer(
					bootstrap_servers=KAFKA_BROKER,
					group_id=GROUP_ID,
					enable_auto_commit=False
				)

				partitions = consumer.partitions_for_topic(TOPIC)

				if partitions is None:
					print("No partitions found")
					return

				topic_partitions = [TopicPartition(TOPIC, p) for p in partitions]

				while True:
					consumer.assign(topic_partitions)

					for tp in topic_partitions:
						committed = consumer.committed(tp)
						consumer.seek_to_end(tp)
						latest = consumer.position(tp)

						lag = latest - (committed or 0)

						consumer_lag.labels(
							topic=tp.topic,
							partition=str(tp.partition)
						).set(lag)

					time.sleep(5)

		8a.3. FULL DEPLOYMENT COMMAND (HARDENED)
		✔️ Production-safe minimal (aligned to your constraints)
			helm install monitoring prometheus-community/kube-prometheus-stack \
			  -n monitoring \
			  --create-namespace \
			  \
			  --set alertmanager.enabled=false \
			  \
			  --set prometheus.prometheusSpec.storageSpec=null \
			  --set prometheus.prometheusSpec.retention=2h \
			  --set prometheus.prometheusSpec.resources.requests.memory=512Mi \
			  --set prometheus.prometheusSpec.resources.limits.memory=1Gi \
			  --set prometheus.prometheusSpec.resources.requests.ephemeral-storage=200Mi \
			  --set prometheus.prometheusSpec.resources.limits.ephemeral-storage=500Mi \
			  \
			  --set grafana.enabled=true \
			  --set grafana.resources.requests.memory=256Mi \
			  --set grafana.resources.limits.memory=512Mi \
			  --set grafana.resources.requests.ephemeral-storage=100Mi \
			  --set grafana.resources.limits.ephemeral-storage=200Mi \
			  \
			  --set kubeStateMetrics.enabled=true \
			  --set nodeExporter.enabled=true \
			  \
			  --set prometheusOperator.resources.requests.memory=128Mi \
			  --set prometheusOperator.resources.limits.memory=256Mi \
			  \
			  --set defaultRules.create=false

		8a.4. WHY THESE SETTINGS (IMPORTANT)
			Prometheus
			Setting				Reason
			storageSpec=null	disables PVC → matches no persistence constraint
			retention=6h		limits memory usage
			512Mi request		fits small node
			1Gi limit			prevents OOM

			Grafana
			Setting				Reason
			256Mi request		minimal UI footprint
			enabled=true		required for visualization

			Component			Reason
