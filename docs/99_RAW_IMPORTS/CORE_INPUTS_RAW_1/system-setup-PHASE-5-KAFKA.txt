		STEP 5.4 — FINAL HELM INSTALL (WORKING CONFIG)
				helm install kafka bitnami/kafka \
				-n kafka \
				--version 26.6.2 \
				--set kraft.enabled=true \
				--set controller.replicaCount=1 \
				--set broker.replicaCount=1 \
				--set persistence.enabled=false \
				--set broker.persistence.enabled=false \
				--set controller.persistence.enabled=false \
				--set volumePermissions.enabled=false \
				--set auth.clientProtocol=plaintext \
				--set listeners.client.protocol=PLAINTEXT \
				--set listeners.controller.protocol=PLAINTEXT \
				--set listeners.interbroker.protocol=PLAINTEXT \
				--set controller.podAntiAffinityPreset=none \
				--set broker.podAntiAffinityPreset=none \
				--set controller.resources.requests.cpu=100m \
				--set controller.resources.requests.memory=256Mi \
				--set broker.resources.requests.cpu=100m \
				--set broker.resources.requests.memory=256Mi \
				--set image.registry=public.ecr.aws \
				--set image.repository=bitnami/kafka \
				--set image.tag=3.6.1

		We will:
			Create Persistent Storage
			Attach to Kafka
			Fix log.dirs

		🔵 STEP 7.1 — DEFINE PVC
			Create file:
			nano kafka-pvc.yaml

				apiVersion: v1
				kind: PersistentVolumeClaim
				metadata:
				  name: kafka-data
				  namespace: kafka
				spec:
				  accessModes:
					- ReadWriteOnce
				  resources:
					requests:
					  storage: 10Gi
				  storageClassName: local-path

			APPLY
				kubectl apply -f kafka-pvc.yaml

			VERIFY
				kubectl get pvc -n kafka

			Expected:
				kafka-data   Bound

		🔵 STEP 7.2 — PATCH KAFKA STATEFULSET

			Now we attach volume to Kafka pod

			GET CURRENT STATEFULSET
				kubectl get statefulset -n kafka

			(usually kafka)

			EDIT
			kubectl edit statefulset kafka -n kafka
			ADD THIS (CRITICAL)
			Under spec.template.spec.containers
			volumeMounts:
			  - name: kafka-data
				mountPath: /var/lib/kafka/data
			Under spec.template.spec
			volumes:
			  - name: kafka-data
				persistentVolumeClaim:
				  claimName: kafka-data

		🔵 STEP 7.3 — FIX log.dirs

			Inside same StatefulSet:

			Find env or config and set:

			KAFKA_LOG_DIRS=/var/lib/kafka/data

			OR if using config:

			log.dirs=/var/lib/kafka/data

			🔴 IMPORTANT

			If Bitnami chart:

			KAFKA_CFG_LOG_DIRS=/bitnami/kafka/data

			Then mount path must match:

			/bitnami/kafka/data

		🔵 STEP 7.4 — RESTART KAFKA
			kubectl delete pod kafka-0 -n kafka
