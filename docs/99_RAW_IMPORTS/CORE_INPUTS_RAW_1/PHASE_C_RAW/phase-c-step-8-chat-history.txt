5.6 Final Code / Functions (CORE)
def generate_output(state_snapshot, orchestration_metadata, version_identifiers):
    instructions = map_state_to_instructions(state_snapshot)

    ordered = sort_instructions(instructions)

    lineage_hash = hash_v1(
        state_snapshot.lineage_hash +
        orchestration_metadata.hash +
        hash_stream(ordered) +
        serialize(version_identifiers)
    )

    output = {
        "execution_instruction_stream": ordered,
        "state_reference": state_snapshot.ref,
        "snapshot_reference": state_snapshot.snapshot,
        "lineage_hash": lineage_hash,
        "orchestration_metadata_reference": orchestration_metadata.ref,
        "version_identifiers": version_identifiers
    }

    output["execution_ready_output_hash"] = hash_v1(serialize(output))

    return output
