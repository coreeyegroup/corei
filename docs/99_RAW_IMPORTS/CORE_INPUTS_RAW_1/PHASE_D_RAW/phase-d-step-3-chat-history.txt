for intent in ordered_intents:

    capital = core_capital if intent.origin == "CORE" else external_capital

    if capital < 0 or exposure_remaining < 0:
        reject(intent)

    approved_size = min(
        intent.requested_size,
        position_limit_remaining,
        capital,
        exposure_remaining
    )

    if hard_constraint_violation:
        reject(intent)

    elif approved_size < intent.requested_size:
        status = "REDUCED"
    else:
        status = "APPROVED"

    # exposure update
    if intent.direction == "BUY":
        exposure += approved_size
    else:
        exposure -= approved_size

    # capital update
    if intent.origin == "CORE":
        core_capital -= approved_size
    else:
        external_capital -= approved_size
