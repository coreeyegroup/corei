5.6 Final Code / Functions
def generate_execution_order(intent):
    timestamp = intent.event_time

    size = (
        intent.modified_size
        if intent.policy_status == "MODIFY"
        else intent.approved_size
    )

    order_id = hash_v1(
        intent.policy_decision_intent_id,
        intent.instrument,
        intent.direction,
        size,
        timestamp
    )

    lineage_hash = hash_v1(
        intent.policy_decision_intent_id,
        intent.risk_approved_intent_id,
        intent.decision_validated_intent_id,
        intent.execution_intent_id,
        intent.trade_intent_id
    )

    return ExecutionOrder(
        order_id=order_id,
        size=size,
        timestamp=timestamp,
        lineage_hash_reference=lineage_hash
    )
