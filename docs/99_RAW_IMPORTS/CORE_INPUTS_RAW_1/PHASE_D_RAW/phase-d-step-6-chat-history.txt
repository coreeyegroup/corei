5.6 Final Code / Functions (CORE)
def generate_events(order, config):
    events = []

    events.append(create_event("ORDER_CREATED"))

    events.append(create_event("ORDER_ROUTED"))

    remaining = order.size

    while remaining > 0:
        size = deterministic_fill(remaining, config)
        remaining -= size

        if remaining == 0:
            events.append(create_event("FULL_FILL", size))
        else:
            events.append(create_event("PARTIAL_FILL", size))

    events.append(create_event("ORDER_COMPLETED"))

    return events
