5.6 Final Code / Functions
def resolve_conflicts(intents):
    sorted_nodes = sort_nodes(intents)

    selected = []
    eliminated = set()

    for node in sorted_nodes:
        if node.id not in eliminated:
            selected.append(node)
            for neighbor in node.conflicts:
                eliminated.add(neighbor.id)

    return selected
