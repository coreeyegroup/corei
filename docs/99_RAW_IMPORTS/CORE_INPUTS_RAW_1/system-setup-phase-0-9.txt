			5.4 LAG CALCULATION FUNCTION

				def calculate_lag():
					consumer = KafkaConsumer(
						bootstrap_servers=KAFKA_BOOTSTRAP,
						group_id=GROUP_ID,
						enable_auto_commit=False
					)

					partitions = consumer.partitions_for_topic(TOPIC)
					topic_partitions = [TopicPartition(TOPIC, p) for p in partitions]

					consumer.assign(topic_partitions)

					end_offsets = consumer.end_offsets(topic_partitions)
					committed_offsets = {
						tp: consumer.committed(tp) for tp in topic_partitions
					}

					total_lag = 0

					for tp in topic_partitions:
						end = end_offsets[tp]
						committed = committed_offsets[tp] or 0
						lag = end - committed
						total_lag += lag

					consumer.close()
					return total_lag

	5. 📉 LAG MONITOR ENGINE (CRITICAL)
		📄 app/lag_monitor.py

			import time
			from kafka import KafkaConsumer, TopicPartition
			from app.metrics import consumer_lag

			KAFKA_BROKER = "kafka.kafka.svc.cluster.local:9092"
			TOPIC = "test-topic"
			GROUP_ID = "metrics-consumer-group"

			def monitor_lag():
				consumer = KafkaConsumer(
					bootstrap_servers=KAFKA_BROKER,
					group_id=GROUP_ID,
					enable_auto_commit=False
				)

				partitions = consumer.partitions_for_topic(TOPIC)

				if partitions is None:
					print("No partitions found")
					return

				topic_partitions = [TopicPartition(TOPIC, p) for p in partitions]

				while True:
					consumer.assign(topic_partitions)

					for tp in topic_partitions:
						committed = consumer.committed(tp)
						consumer.seek_to_end(tp)
						latest = consumer.position(tp)

						lag = latest - (committed or 0)

						consumer_lag.labels(
							topic=tp.topic,
							partition=str(tp.partition)
						).set(lag)

					time.sleep(5)

				set -e

				REGISTRY="localhost:5000/core"
				VERSION_FILE=~/core-system/platform/versioning/version.txt

				get_version() {
				  SERVICE=$1
				  grep "^$SERVICE=" $VERSION_FILE | cut -d'=' -f2
				}

				SERVICES=(
				  "signal-service"
				)

				for SERVICE in "${SERVICES[@]}"
				do
				  VERSION=$(get_version $SERVICE)

				  if [ -z "$VERSION" ]; then
					echo "ERROR: Version not found for $SERVICE"
					exit 1
				  fi

				  echo "Checking existing versions..."

				  if curl -s http://localhost:5000/v2/core/$SERVICE/tags/list | grep -q "$VERSION"; then
					echo "ERROR: Version $VERSION already exists for $SERVICE"
					exit 1
				  fi

				  echo "--------------------------------------"
				  echo "Building $SERVICE:$VERSION"
				  echo "--------------------------------------"

				  docker build --no-cache -t $REGISTRY/$SERVICE:$VERSION \
					~/core-system/services/$SERVICE

				  echo "Pushing $SERVICE:$VERSION"

				  docker push $REGISTRY/$SERVICE:$VERSION

				done

				echo "--------------------------------------"
				echo "VERSIONED BUILD COMPLETE"
