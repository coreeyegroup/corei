        (WorkspaceRegistry as WorkspaceModel[]).push(

        const exists =

        if (!exists) {

            throw new Error(

            );

        }

        this.activeWorkspaceId =
            workspaceId;

    }

    public getActive():
        | WorkspaceModel

        return WorkspaceRegistry.find(
            workspace =>
