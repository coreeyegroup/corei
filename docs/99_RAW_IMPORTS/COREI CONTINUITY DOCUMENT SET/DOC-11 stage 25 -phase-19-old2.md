import type { WorkspaceContract } from "../../workspace/core/contracts/workspace-contract";

export class WorkspaceOperationalCoordinator
implements WorkspaceOperationalContract {

    private readonly workspaces = new Map<string, WorkspaceContract>();

    public activeWorkspaceId: string | null = null;

    async create(workspace: WorkspaceContract): Promise<void> {
        this.workspaces.set(workspace.id, workspace);
    }

    async open(workspaceId: string): Promise<void> {
        this.activeWorkspaceId = workspaceId;
    }

    async activate(workspaceId: string): Promise<void> {
        this.activeWorkspaceId = workspaceId;
    }

    async deactivate(_: string): Promise<void> {
        this.activeWorkspaceId = null;
    }

    async suspend(_: string): Promise<void> {}

    async resume(workspaceId: string): Promise<void> {
        this.activeWorkspaceId = workspaceId;
    }

    async close(workspaceId: string): Promise<void> {
        if (this.activeWorkspaceId === workspaceId) {
            this.activeWorkspaceId = null;
        }
    }

    async dispose(workspaceId: string): Promise<void> {
        this.workspaces.delete(workspaceId);

        if (this.activeWorkspaceId === workspaceId) {
            this.activeWorkspaceId = null;
        }
    }

    exists(workspaceId: string): boolean {
        return this.workspaces.has(workspaceId);
    }

    getActiveWorkspace(): WorkspaceContract | null {
        return this.activeWorkspaceId
            ? this.workspaces.get(this.activeWorkspaceId) ?? null
            : null;
    }

    getWorkspace(workspaceId: string): WorkspaceContract | null {
        return this.workspaces.get(workspaceId) ?? null;
    }

    getWorkspaces(): readonly WorkspaceContract[] {
        return [...this.workspaces.values()];
    }
}
